cmake_minimum_required(VERSION 3.20.0)

set(CONFIG_APPLICATION_DEFINED_SYSCALL true)
option(YUPAD43_TEST_HORIZONTAL_SWIPE "Exercise two-finger horizontal gesture events" OFF)
option(YUPAD43_TEST_NO_TAPDRAG "Disable tap/recontact dragging" OFF)
option(YUPAD43_TEST_TRANSPORT "Exercise the real driver with emulated I2C and RDY" OFF)
if(YUPAD43_TEST_TRANSPORT)
    list(APPEND DTS_ROOT ${CMAKE_CURRENT_LIST_DIR}/../..)
    set(DTC_OVERLAY_FILE ${CMAKE_CURRENT_LIST_DIR}/transport.overlay)
    list(APPEND EXTRA_CONF_FILE ${CMAKE_CURRENT_LIST_DIR}/transport.conf)
endif()

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(iqs9151_work_cb_test)

target_sources(app PRIVATE ../../drivers/input/iqs9151.c)
if(YUPAD43_TEST_TRANSPORT)
    target_sources(app PRIVATE src/transport.c)
elseif(YUPAD43_TEST_HORIZONTAL_SWIPE)
    target_sources(app PRIVATE src/horizontal_swipe.c)
else()
    target_sources(app PRIVATE src/main.c)
endif()

target_include_directories(app PRIVATE
    ../../drivers/input
    ../../include
)

option(YUPAD43_TEST_CUSTOM_CONFIG "Exercise common non-default gesture settings" OFF)
if(YUPAD43_TEST_CUSTOM_CONFIG)
    target_compile_definitions(app PRIVATE
        YUPAD43_TEST_CUSTOM_CONFIG=1
        CONFIG_INPUT_IQS9151_3F_SWIPE_THRESHOLD=0
    )
else()
    target_compile_definitions(app PRIVATE
        CONFIG_INPUT_IQS9151_1F_TAP_ENABLE=1
        CONFIG_INPUT_IQS9151_2F_TAP_ENABLE=1
        CONFIG_INPUT_IQS9151_3F_TAP_ENABLE=1
        CONFIG_INPUT_IQS9151_3F_SWIPE_THRESHOLD=300
    )
endif()

if(YUPAD43_TEST_HORIZONTAL_SWIPE)
    target_compile_definitions(app PRIVATE CONFIG_INPUT_IQS9151_2F_HORIZONTAL_SWIPE=1)
elseif(YUPAD43_TEST_CUSTOM_CONFIG)
    target_compile_definitions(app PRIVATE CONFIG_INPUT_IQS9151_2F_HORIZONTAL_DISABLED=1)
else()
    target_compile_definitions(app PRIVATE CONFIG_INPUT_IQS9151_2F_HORIZONTAL_SCROLL=1)
endif()

if(NOT YUPAD43_TEST_NO_TAPDRAG)
    target_compile_definitions(app PRIVATE
        CONFIG_INPUT_IQS9151_1F_PRESSHOLD_ENABLE=1
        CONFIG_INPUT_IQS9151_2F_PRESSHOLD_ENABLE=1
        CONFIG_INPUT_IQS9151_3F_PRESSHOLD_ENABLE=1
    )
endif()

    CONFIG_INPUT_IQS9151_CURSOR_INERTIA_DECAY=970
    CONFIG_INPUT_IQS9151_CURSOR_INERTIA_RECENT_WINDOW_MS=60
    CONFIG_INPUT_IQS9151_CURSOR_INERTIA_STALE_GAP_MS=35
    CONFIG_INPUT_IQS9151_CURSOR_INERTIA_MIN_SAMPLES=2
    CONFIG_INPUT_IQS9151_CURSOR_INERTIA_MIN_AVG_SPEED=10
    CONFIG_INPUT_IQS9151_1F_TAP_MAX_MS=120
